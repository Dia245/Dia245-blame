// Opciones de geolocalización
const GEO_OPTIONS = {
  enableHighAccuracy: true,
  timeout: 15000,
  maximumAge: 0
};

// Iniciar rastreo continuo
function startTracking() {
  if (!navigator.geolocation) return;

  watchId = navigator.geolocation
    .watchPosition(onSuccess, onError, GEO_OPTIONS);
}

// Callback con datos de posición
function onSuccess(pos) {
  const {
    latitude, longitude,
    accuracy, altitude,
    speed, heading
  } = pos.coords;

  // Mover marcador en el mapa
  marker.setLatLng([latitude, longitude]);

  // Construir payload para la API
  const payload = {
    device_id: "device_001",
    timestamp: new Date(pos.timestamp).toISOString(),
    location: {
      latitude,
      longitude,
      accuracy_meters: accuracy,
      altitude_meters: altitude,
      speed_kmh: speed ? speed * 3.6 : null,
      heading_degrees: heading
    }
  };

  // Enviar a la API REST
  sendToAPI(payload);
}

// Envío a API propia
async function sendToAPI(payload) {
  await fetch('/api/v1/locations', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${TOKEN}`
    },
    body: JSON.stringify(payload)
  });
}

// Detener rastreo
function stopTracking() {
  navigator.geolocation
    .clearWatch(watchId);
}
# Endpoint REST — Node.js / Express

const express = require('express');
const app = express();
app.use(express.json());

// POST /api/v1/locations
app.post('/api/v1/locations',
  authMiddleware,
  async (req, res) => {

  const {
    device_id, timestamp, location
  } = req.body;

  // Validar campos requeridos
  if (!device_id || !location?.latitude) {
    return res.status(400).json({
      error: 'device_id y location requeridos'
    });
  }

  // Guardar en base de datos
  await db.query(
    `INSERT INTO locations
     (device_id, latitude, longitude,
      accuracy, altitude, speed,
      heading, recorded_at)
     VALUES ($1,$2,$3,$4,$5,$6,$7,$8)`,
    [
      device_id,
      location.latitude,
      location.longitude,
      location.accuracy_meters,
      location.altitude_meters,
      location.speed_kmh,
      location.heading_degrees,
      timestamp || new Date()
    ]
  );

  res.json({
    status: 'ok',
    received_at: new Date().toISOString()
  });
});

// GET — historial por device_id
app.get('/api/v1/locations/:device_id',
  async (req, res) => {
  const rows = await db.query(
    `SELECT * FROM locations
     WHERE device_id = $1
     ORDER BY recorded_at DESC
     LIMIT 1000`,
    [req.params.device_id]
  );
  res.json(rows.rows);
});

app.listen(3000);
-- PostgreSQL + PostGIS Schema

CREATE EXTENSION IF NOT EXISTS postgis;

-- Dispositivos registrados
CREATE TABLE devices (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name       TEXT NOT NULL,
  user_id    UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabla principal de ubicaciones
CREATE TABLE locations (
  id          BIGSERIAL PRIMARY KEY,
  device_id   UUID REFERENCES devices(id),

  -- Coordenadas
  latitude    DOUBLE PRECISION NOT NULL,
  longitude   DOUBLE PRECISION NOT NULL,
  geom        GEOMETRY(Point, 4326),

  -- Datos del sensor
  accuracy    FLOAT,
  altitude    FLOAT,
  speed       FLOAT,
  heading     FLOAT,

  -- Timestamps
  recorded_at TIMESTAMPTZ NOT NULL,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Auto-calcular columna geom desde lat/lng
CREATE OR REPLACE FUNCTION set_geom()
RETURNS TRIGGER AS $$
BEGIN
  NEW.geom := ST_SetSRID(
    ST_MakePoint(NEW.longitude, NEW.latitude),
    4326
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_geom
  BEFORE INSERT OR UPDATE ON locations
  FOR EACH ROW EXECUTE FUNCTION set_geom();

-- Índices para consultas rápidas
CREATE INDEX idx_loc_device
  ON locations(device_id, recorded_at DESC);
CREATE INDEX idx_loc_geom
  ON locations USING GIST(geom);

-- Consulta: ruta de un device hoy
SELECT latitude, longitude,
       speed, recorded_at
FROM   locations
WHERE  device_id = 'uuid-aqui'
  AND  recorded_at >= NOW() - INTERVAL '1 day'
ORDER BY recorded_at;
